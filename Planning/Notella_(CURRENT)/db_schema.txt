// Notella - Student Notes & Flashcard Platform
// Docs: https://dbml.dbdiagram.io/docs

Table users {
  user_id integer [primary key]
  username varchar [not null, unique]
  email varchar [not null, unique]
  password_hash varchar [not null]
  pfp_filepath varchar [default: `NULL`, note: 'If null, no profile picture has been set']
  user_tier varchar [not null, default: 'free', note: 'free or paid']
  created_at timestamp [default: `now()`]
  light_mode boolean [not null, default:false]
}

Table notes {
  note_id integer [primary key]
  user_id integer [not null]
  copied_from integer [note: 'null if original, points to original note id if copied']
  title varchar [not null]
  content_md text [note: 'Markdown content']
  description text
  is_public boolean [not null, default: false]
  created_at timestamp [default: `now()`]
  accessed_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table note_likes {
  user_id int [not null]
  note_id int [not null]

  indexes {
    (user_id, note_id) [pk]
  }
}

Table tags {
  tag_id integer [primary key]
  name varchar [not null, unique]
}

Table note_tags {
  note_id integer [not null]
  tag_id integer [not null]

  indexes {
    (note_id, tag_id) [pk]
  }
}

Table decks {
  deck_id integer [primary key]
  user_id integer [not null]
  copied_from integer [note: 'null if original, points to original deck id if copied']
  title varchar [not null]
  is_public boolean [not null, default: false]
  created_at timestamp [default: `now()`]
  accessed_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table deck_tags {
  deck_id integer [not null]
  tag_id integer [not null]

  indexes {
    (deck_id, tag_id) [pk]
  }
}

Table deck_likes {
  user_id int [not null]
  deck_id int [not null]

  indexes {
    (user_id, deck_id) [pk]
  }
}

Table flashcards {
  flashcard_id integer [primary key]
  deck_id integer [not null]
  front text [not null]
  back text [not null]
  order_index integer [not null, default: 0]
}

Table quizzes {
  quiz_id integer [primary key]
  note_id integer [not null]
  name varchar [not null]
  total_questions integer [not null, default: 0]
  total_correct integer [not null, default: 0]
  created_at timestamp [default: `now()`]
  last_accessed timestamp [default: `now()`]
}

Table quiz_questions {
  question_id integer [primary key]
  quiz_id integer [not null]
  question_type varchar [not null, default: 'mcq', note: 'mcq or written']
  question_text text [not null]
  option_a varchar [note: 'null if question_type is written']
  option_b varchar [note: 'null if question_type is written']
  option_c varchar [note: 'null if question_type is written']
  option_d varchar [note: 'null if question_type is written']
  correct_answer varchar [not null, note: 'a, b, c, or d for mcq, text for written']
  user_answer varchar [note: 'null if not answered']
  order_index integer [not null, default: 0]
}

Table flashcard_results {
  results_id           integer   [primary key]
  user_id      integer   [not null]
  flashcard_id integer   [not null]
  is_correct   boolean   [not null]
  attempted_at timestamp [default: `now()`]

  indexes {
    (user_id, flashcard_id) [unique, note: 'one result per user per card — upsert on conflict']
  }
}

Table deck_progress {
  progress_id integer [pk]
  user_id integer [not null]
  deck_id integer [not null]
  current_index integer [not null]
  updated_at timestamp
}

Table password_resets {
  pw_reset_id integer [primary key]
  user_id integer [not null]
  token_hash text [not null]
  expires_at timestamp
  used_at timestamp
}

Ref: password_resets.user_id      > users.user_id
Ref: flashcard_results.user_id      > users.user_id
Ref: flashcard_results.flashcard_id > flashcards.flashcard_id

// --- Relationships ---

// Users own notes and decks
Ref user_notes: notes.user_id > users.user_id
Ref user_decks: decks.user_id > users.user_id

// Notes and decks can be copied from other notes/decks
Ref note_copied_from: notes.copied_from > notes.note_id
Ref deck_copied_from: decks.copied_from > decks.deck_id

// Tags applied to notes and decks
Ref: note_tags.note_id > notes.note_id
Ref: note_tags.tag_id > tags.tag_id
Ref: deck_tags.deck_id > decks.deck_id
Ref: deck_tags.tag_id > tags.tag_id

// Each user can like notes and decks
ref: note_likes.user_id > users.user_id
ref: note_likes.note_id > notes.note_id
ref: deck_likes.user_id > users.user_id
ref: deck_likes.deck_id > decks.deck_id

// Decks contain flashcards
Ref deck_flashcards: flashcards.deck_id > decks.deck_id

// Decks have progress
Ref: decks.deck_id < deck_progress.deck_id

// Notes generate quiz sessions
Ref note_quiz: quizzes.note_id > notes.note_id

// Quizzes contain questions
Ref quiz_questions_ref: quiz_questions.quiz_id > quizzes.quiz_id
